async function register(){

 let n = rn.value.trim();
 let a = ru.value.trim().toLowerCase();
 let p = rp.value;
 let p2 = rp2.value;
 let e = re.value.trim().toLowerCase();
 let invite = ric.value.trim();

 if(invite !== db().inviteCode)
   return alert('Geçersiz HEDEFİM Davet Kodu.');

 if(!n || !a || !p || !p2 || !e || !invite)
   return alert('Tüm alanları doldur.');

 if(p.length < 6)
   return alert('Şifre en az 6 karakter olmalı.');

 if(p !== p2)
   return alert('Şifreler aynı değil.');

 let x = db();

 if(x.users[a])
   return alert('Bu kullanıcı adı zaten kayıtlı.');

 let r = prompt('Şifre kurtarma cevabın:');

 if(!r)
   return;

 try{

   let cred = await window.firebaseCreateUser(
     window.firebaseAuth,
     e,
     p
   );

   x.users[a] = {

     name:n,
     username:a,
     email:e,
     uid:cred.user.uid,

     recovery:r.toLowerCase().trim(),

     progress:{},
     quiz:{},
     custom:[],
     messages:[],
     sent:[],
     notes:[],
     advice:[],
     announcements:[],
     programs:[],
     surprises:true

   };

   x.session = a;

   save(x);

   alert('✅ Kayıt başarılı!');

   open();

 }catch(err){

   if(err.code === 'auth/email-already-in-use')
     return alert('Bu e-posta zaten kayıtlı.');

   if(err.code === 'auth/invalid-email')
     return alert('Geçerli bir e-posta adresi gir.');

   if(err.code === 'auth/weak-password')
     return alert('Şifre çok zayıf. En az 6 karakter kullan.');

   console.error(err);

   alert('Kayıt sırasında bir hata oluştu.');

 }

}

function login(){

 let a = lu.value.trim().toLowerCase();
 let p = lp.value;

 let x = db();

 if(!a || !p)
   return alert('Kullanıcı adı ve şifreyi gir.');

 let user = x.users[a];

 if(!user)
   return alert('Kullanıcı adı veya şifre hatalı.');

 if(!user.email)
   return alert('Bu hesap Firebase hesabına bağlı değil.');

 window.firebaseSignIn(
   window.firebaseAuth,
   user.email,
   p
 )
 .then(()=>{

   x.session = a;
   save(x);
   open();

 })
 .catch(err=>{

   if(
     err.code === 'auth/invalid-credential' ||
     err.code === 'auth/wrong-password' ||
     err.code === 'auth/user-not-found'
   ){
     return alert('Kullanıcı adı veya şifre hatalı.');
   }

   console.error(err);
   alert('Giriş sırasında bir hata oluştu.');

 });

}

let selectedSound='pop';

function playSurpriseSound(){

 try{

  const AudioContext=
   window.AudioContext||
   window.webkitAudioContext;

  const ctx=new AudioContext();

  const osc=ctx.createOscillator();
  const gain=ctx.createGain();

  osc.connect(gain);
  gain.connect(ctx.destination);

  osc.frequency.value=
   selectedSound==='bell'?700:
   selectedSound==='fun'?420:
   520;

  gain.gain.setValueAtTime(.08,ctx.currentTime);

  osc.start();

  osc.frequency.exponentialRampToValueAtTime(
   800,
   ctx.currentTime+.15
  );

  osc.stop(ctx.currentTime+.18);

 }catch(e){}
}

function addTopic(){

 let t=nt.value.trim();

 let x=db();
 let z=u();

 if(!t)return alert('Konu adını yaz.');

 z.custom??=[];

 let exists=
 z.custom.some(a=>a.s===ns.value&&a.t.toLowerCase()===t.toLowerCase());

 if(exists)return alert('Bu konu zaten eklenmiş.');

 z.custom.push({
  s:ns.value,

 });

}
